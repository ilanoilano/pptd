python -c "
import csv

sequences = []
with open('results/6Y9A/energies.csv', 'r') as f:
    reader = csv.reader(f)
    next(reader, None)  # 跳过表头
    for row in reader:
        if row:
            seq = row[0].strip()
            if seq:
                sequences.append(seq)

with open('sequences.txt', 'w') as f:
    for seq in sequences:
        f.write(f'{seq}\n')

print(f'已写入 {len(sequences)} 个序列到 sequences.txt')
"
